{{- $folder := .Get "folder" -}} {{- $nbColumns := 2 -}} {{- $images := resources.Match (printf "%s/*" $folder) -}}
{{- range $index, $image := $images -}} {{/* ------------------------------------------------------------ Corriger l'orientation EXIF avant toute autre opération ------------------------------------------------------------ */}} {{- $image = $image.Filter images.AutoOrient -}} {{- $width := $image.Width -}} {{- $height := $image.Height -}} {{/* ------------------------------------------------------------ Génération des différentes tailles sans agrandissement ------------------------------------------------------------ */}} {{- $srcset := slice -}} {{- if ge $width 480 -}} {{- $img480 := $image.Resize "480x webp" -}} {{- $srcset = $srcset | append (printf "%s 480w" $img480.RelPermalink) -}} {{- end -}} {{- if ge $width 768 -}} {{- $img768 := $image.Resize "768x webp" -}} {{- $srcset = $srcset | append (printf "%s 768w" $img768.RelPermalink) -}} {{- end -}} {{- if ge $width 1024 -}} {{- $img1024 := $image.Resize "1024x webp" -}} {{- $srcset = $srcset | append (printf "%s 1024w" $img1024.RelPermalink) -}} {{- end -}} {{- if ge $width 1280 -}} {{- $img1280 := $image.Resize "1280x webp" -}} {{- $srcset = $srcset | append (printf "%s 1280w" $img1280.RelPermalink) -}} {{- end -}} {{/* ------------------------------------------------------------ Image par défaut ------------------------------------------------------------ */}} {{- $src := "" -}} {{- if ge $width 1280 -}} {{- $src = ($image.Resize "1280x webp").RelPermalink -}} {{- else if ge $width 1024 -}} {{- $src = ($image.Resize "1024x webp").RelPermalink -}} {{- else if ge $width 768 -}} {{- $src = ($image.Resize "768x webp").RelPermalink -}} {{- else if ge $width 480 -}} {{- $src = ($image.Resize "480x webp").RelPermalink -}} {{- else -}} {{- $src = ($image.Process "webp").RelPermalink -}} {{- end -}} {{/* ------------------------------------------------------------ Alternance gauche / droite ------------------------------------------------------------ */}}
{{- end -}}
